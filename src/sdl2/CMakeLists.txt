cmake_minimum_required(VERSION 3.20)

project(GamebuinoClassicSDL2 LANGUAGES C)

# -----------------------------------------------------------------------------
# The SDL2 port - a standalone CMake project in its own right, not a
# subdirectory of a repo-wide top-level CMakeLists.txt (there isn't one -
# mirrors src/sdl3/CMakeLists.txt's own layout exactly, one fully
# self-contained CMakeLists.txt per port, each `cd`'d into and built on its
# own). Build from here:
#
#   cd src/sdl2
#   cmake -B build -G Ninja
#   cmake --build build
#
# cmake_minimum_required is 3.20 here, not src/sdl3's 3.5 - matches the
# sibling Tinyjoypad_SDL project's own src/sdl2/CMakeLists.txt exactly;
# SDL2's own CMake config needs the newer minimum, SDL3's own doesn't.
#
# Pulls in the shared, port-agnostic "game world" from ../gameworld (see
# that directory's own machineDependent.h for the interface boundary every
# port implements) - unchanged from src/sdl3's own identical setup.
# -----------------------------------------------------------------------------

# Create an option to switch between a system SDL2 install and a vendored
# SDL2 source checkout - matches src/sdl3/CMakeLists.txt's own
# USE_VENDORED_SDL option and the sibling Tinyjoypad_SDL project's own
# src/sdl2/CMakeLists.txt.
option(USE_VENDORED_SDL "Use a vendored SDL2 source checkout instead of the system package" OFF)
set(SDL_VENDOR_DIR "${PROJECT_SOURCE_DIR}/../../SDL2" CACHE PATH "Path to a vendored SDL2 source checkout (only used if USE_VENDORED_SDL is on)")

if(USE_VENDORED_SDL)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC_PIC ON CACHE BOOL "" FORCE)
    add_subdirectory("${SDL_VENDOR_DIR}" "${PROJECT_SOURCE_DIR}/build/SDL2" EXCLUDE_FROM_ALL)
else()
    # 1. Look for an SDL2 package, 2. look for the SDL2 component, 3. fail if none can be found
    find_package(SDL2 REQUIRED CONFIG REQUIRED COMPONENTS SDL2)

    # 1. Look for an SDL2 package, 2. look for the SDL2main component, 3. DO NOT fail when SDL2main is not available
    find_package(SDL2 REQUIRED CONFIG COMPONENTS SDL2main)
endif()

# Every shim/menu/games/*.c file is its own separate translation unit (see
# CLAUDE.md's "Translation-unit boundary" section) - gamesMain.c doesn't
# #include any of their .c bodies, so ../gameworld/*.c and
# ../gameworld/games/*.c are globbed directly here alongside this port's own
# *.c, each compiled and linked independently like any normal multi-file C
# project. ../gameworld is the ENTIRE reason this is a multi-port-ready
# layout - it never #includes SDL.h and knows nothing about this port.
file(GLOB SRC CONFIGURE_DEPENDS
        "${PROJECT_SOURCE_DIR}/*.c"
        "${PROJECT_SOURCE_DIR}/*.h"
        "${PROJECT_SOURCE_DIR}/../gameworld/*.c"
        "${PROJECT_SOURCE_DIR}/../gameworld/*.h"
        "${PROJECT_SOURCE_DIR}/../gameworld/games/*.c"
        "${PROJECT_SOURCE_DIR}/../gameworld/games/*.h"
)

if (WIN32)
	set(CMAKE_EXE_LINKER_FLAGS "-static -mwindows")
endif()

add_executable(${PROJECT_NAME})
target_sources(${PROJECT_NAME} PRIVATE ${SRC} )

target_include_directories(${PROJECT_NAME} PRIVATE
    "${PROJECT_SOURCE_DIR}"
    "${PROJECT_SOURCE_DIR}/../gameworld"
    "${PROJECT_SOURCE_DIR}/../../assets/thumbnails"
)

# SDL2 (unlike SDL3) still typically needs SDL2main linked ahead of the
# app's own main() on some platforms/link setups (it supplies a real
# WinMain()-to-main() shim on Windows, among other things) - link it
# whenever the package actually provides it, matching the sibling
# Tinyjoypad_SDL project's own src/sdl2/CMakeLists.txt.
if(TARGET SDL2::SDL2main)
    target_link_libraries(${PROJECT_NAME} PRIVATE SDL2::SDL2main)
endif()

if(USE_VENDORED_SDL)
    target_link_libraries(${PROJECT_NAME} PRIVATE
        SDL2::SDL2-static
    )
else()
    target_link_libraries(${PROJECT_NAME} PRIVATE
        SDL2::SDL2
    )
endif()
