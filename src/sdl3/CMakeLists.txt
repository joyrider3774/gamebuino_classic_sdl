cmake_minimum_required(VERSION 3.5)

project(GamebuinoClassicSDL3 LANGUAGES C)

# -----------------------------------------------------------------------------
# The SDL3 port - a standalone CMake project in its own right, not a
# subdirectory of a repo-wide top-level CMakeLists.txt (there isn't one -
# mirrors the sibling Tinyjoypad_SDL project's own src/sdl3/, src/sdl2/,
# src/playdate/ layout exactly, one fully self-contained CMakeLists.txt per
# port, each `cd`'d into and built on its own). Build from here:
#
#   cd src/sdl3
#   cmake -B build -G Ninja
#   cmake --build build
#
# Pulls in the shared, port-agnostic "game world" from ../gameworld (see
# that directory's own machineDependent.h for the interface boundary every
# port implements) - a future src/sdl2/ or src/playdate/ port reuses that
# same directory unchanged, adding only its own platform-specific files
# alongside its own CMakeLists.txt at that same src/<port>/ level.
#
# Unlike the sibling Tinyjoypad_SDL project (which vendors a full SDL3
# checkout by default), this project builds against the system-installed
# SDL3 package by default - both SDL2 and SDL3 are already available
# system-wide in this project's own build environment, so there was no need
# to vendor a full SDL source tree. USE_VENDORED_SDL is still offered, for
# parity with that project's own option and for anyone who'd rather build
# from a checked-out SDL3 source tree instead - point it at one via
# SDL_VENDOR_DIR (default "../../SDL3", the same relative location that
# project's own vendored checkout lives at) if you use it; this repo does
# not ship one itself.
# -----------------------------------------------------------------------------

option(USE_VENDORED_SDL "Use a vendored SDL3 source checkout instead of the system package" OFF)
set(SDL_VENDOR_DIR "${PROJECT_SOURCE_DIR}/../../SDL3" CACHE PATH "Path to a vendored SDL3 source checkout (only used if USE_VENDORED_SDL is on)")

if(USE_VENDORED_SDL)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC_PIC ON CACHE BOOL "" FORCE)
    add_subdirectory("${SDL_VENDOR_DIR}" "${PROJECT_SOURCE_DIR}/build/SDL3" EXCLUDE_FROM_ALL)
else()
    # 1. Look for an SDL3 package, 2. look for the SDL3 component, 3. fail if none can be found
    find_package(SDL3 REQUIRED CONFIG REQUIRED COMPONENTS SDL3)
endif()

# Every shim/menu/games/*.c file is its own separate translation unit (see
# CLAUDE.md's "Translation-unit boundary" section) - gamesMain.c doesn't
# #include any of their .c bodies, so ../gameworld/*.c and
# ../gameworld/games/*.c are globbed directly here alongside this port's own
# *.c, each compiled and linked independently like any normal multi-file C
# project. ../gameworld is the ENTIRE reason this is a multi-port-ready
# layout - it never #includes SDL.h and knows nothing about this port.
file(GLOB SRC CONFIGURE_DEPENDS
        "${PROJECT_SOURCE_DIR}/*.c"
        "${PROJECT_SOURCE_DIR}/*.h"
        "${PROJECT_SOURCE_DIR}/../gameworld/*.c"
        "${PROJECT_SOURCE_DIR}/../gameworld/*.h"
        "${PROJECT_SOURCE_DIR}/../gameworld/games/*.c"
        "${PROJECT_SOURCE_DIR}/../gameworld/games/*.h"
)

if (WIN32)
	set(CMAKE_EXE_LINKER_FLAGS "-static -mwindows")
endif()

add_executable(${PROJECT_NAME})
target_sources(${PROJECT_NAME} PRIVATE ${SRC} )

target_include_directories(${PROJECT_NAME} PRIVATE
    "${PROJECT_SOURCE_DIR}"
    "${PROJECT_SOURCE_DIR}/../gameworld"
    "${PROJECT_SOURCE_DIR}/../../assets/thumbnails"
)

if(USE_VENDORED_SDL)
    target_link_libraries(${PROJECT_NAME} PRIVATE
        SDL3::SDL3-static
    )
else()
    target_link_libraries(${PROJECT_NAME} PRIVATE
        SDL3::SDL3
    )
endif()
